algoritmo "L04H"

var
  nome, continuar: caractere
  largura, comprimento, area, totalArea: inteiro

inicio

  totalArea <- 0

  repita
  escreval("Nome do comodo: ")
  leia(nome)
  escreval("Largura: ")
  leia(largura)
  escreval("Comprimento: ")
  leia(comprimento)

  area <- largura * comprimento

  escreval("Area do ", nome, " : ", area, "m2")

  totalArea <- totalArea + area

  escreval("DESEJA CONTINUAR (SIM/NAO)")
  leia(continuar)
  ate continuar = "NAO"

  escreval("Area total: ", totalArea, "m2")
fimalgoritmo

algoritmo "L04J"

var
   dividendo, divisor, quociente: inteiro

inicio

   escreval("Digite o dividendo: ")
   leia(dividendo)

   escreval("Digite o divisor: ")
   leia(divisor)

   quociente <- 0

   repita
      dividendo <- dividendo - divisor
      quociente <- quociente + 1
   ate dividendo < divisor

   escreval("Quociente: ", quociente)

fimalgoritmo
