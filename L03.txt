algoritmo "L03K"

var
  nomeComodo, continuar: caractere
  area, largura, comprimento, totalArea: real

inicio

  continuar <- "sim"
  totalArea <- 0

  enquanto continuar <> "nao" faca
   escreval("------------------------------")
   escreval("Programa para calcular a area")
   escreval("------------------------------")

   escreval("Qual o nome do comodo?")
   leia(nomeComodo)
   escreval("Qual a largura do comodo?")
   leia(largura)
   escreval("Qual é o comprimento do comodo?")
   leia(comprimento)
   escreval("O nome do comodo é ", nomeComodo, " ele tem a seguinte largura ", largura, " e o seguinte cumprimento ", comprimento )
   area <- largura * comprimento
   escreval("tamanho da area deste comodo ", area)


   escreval("--------------------")
   escreval("Tem mais algum comodo digite 'sim' caso contrario digite 'nao'")
   leia(continuar)

   totalArea <- totalArea + area
  
  fimenquanto

  escreval("A Area total da sua residencia é ", totalArea)

algoritmo "L03L"

var
   valorDigitado, maior, menor: inteiro

inicio

   escreval("Digite valores positivos. Um valor negativo encerra o programa.")
   leia(valorDigitado)

   se valorDigitado >= 0 entao
      maior <- valorDigitado
      menor <- valorDigitado

      enquanto valorDigitado >= 0 faca

         escreval("O valor que voce digitou foi: ", valorDigitado)

         se valorDigitado > maior entao
            maior <- valorDigitado
         fimse

         se valorDigitado < menor entao
            menor <- valorDigitado
         fimse

         leia(valorDigitado)

      fimenquanto

      escreval("O maior valor digitado foi: ", maior)
      escreval("O menor valor digitado foi: ", menor)

   senao
      escreval("Nenhum valor positivo foi informado.")

   fimse

fimalgoritmo
