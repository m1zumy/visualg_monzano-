algoritmo "L05H"

   expoente, base, i, resultado: inteiro

inicio

   escreva("Digite o expoente: ")
   leia(expoente)

   escreva("Coloque a base: ")
   leia(base)

   resultado <- 1

   para i de 1 ate expoente faca
      resultado <- resultado * base
   fimpara

   escreval("A base ", base, " com o expoente ", expoente, " = ", resultado)

algoritmo "L05I"

var
   anterior, atual, proximo, i: inteiro

inicio

   anterior <- 1
   atual <- 1

   para i de 1 ate 15 faca
      escreval(atual)

      proximo <- anterior + atual
      anterior <- atual
      atual <- proximo
   fimpara

algoritmo "L05K"

var
   fatorial, i, j: inteiro

inicio

   fatorial <- 1

   para i de 1 ate 10 faca
      se i % 2 = 1 entao
         fatorial <- 1

         para j de 1 ate i faca
            fatorial <- fatorial * j
         fimpara

         escreval(fatorial)
      fimse
   fimpara
